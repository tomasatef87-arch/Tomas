<script>
    const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbzChTXABZ_gCoa1mhWNS3SHiMsnmy_1lduY2alJ2VnE0aJhOPAxrvahnLntcDEGWxZQ/exec";

    // دالة الإرسال عبر GET لضمان وصول البيانات دون حظر من المتصفح
    async function syncDataToSheet(sheetName, rowData) {
        try {
            let encodedData = encodeURIComponent(JSON.stringify(rowData));
            let url = `${WEB_APP_URL}?sheetName=${sheetName}&data=${encodedData}`;
            let response = await fetch(url);
            let result = await response.json();
            return result;
        } catch (error) {
            console.error("خطأ في المزامنة:", error);
        }
    }

    function checkLogin() {
        let u = document.getElementById('loginUser').value.trim();
        let p = document.getElementById('loginPass').value.trim();
        if (u === 'tomas' && p === '1234') {
            document.getElementById('loginOverlay').style.display = 'none';
            document.getElementById('mainAppContainer').style.display = 'block';
            sessionStorage.setItem('isLoggedIn', 'true');
            initApp();
        } else {
            document.getElementById('loginError').style.display = 'block';
        }
    }

    function logout() {
        sessionStorage.removeItem('isLoggedIn');
        location.reload();
    }

    window.onload = function() {
        if (sessionStorage.getItem('isLoggedIn') === 'true') {
            document.getElementById('loginOverlay').style.display = 'none';
            document.getElementById('mainAppContainer').style.display = 'block';
            initApp();
        }
    };

    function initApp() {
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('cDate').value = today;
        document.getElementById('expDate').value = today;
    }

    function showTab(tabId, btn) {
        document.querySelectorAll('.tab-panel').forEach(p => { p.classList.remove('active'); p.style.display = 'none'; });
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        let selected = document.getElementById(tabId);
        if(selected) { selected.classList.add('active'); selected.style.display = 'block'; }
        if(btn) btn.classList.add('active');
    }

    function saveCash() {
        let date = document.getElementById('cDate').value;
        let act = parseFloat(document.getElementById('cActual').value);
        let sys = parseFloat(document.getElementById('cSystem').value);
        if (isNaN(act) || isNaN(sys)) return alert('أدخل القيم الصحيحة');

        let diff = act - sys;
        let st = diff === 0 ? 'متطابق' : (diff > 0 ? `زيادة (+${diff.toFixed(2)})` : `عجز (${diff.toFixed(2)})`);

        syncDataToSheet('Records', [date, 'خزنة نقدية', 'نقدي', 'تقفيل درج', `الفعلي: ${act} | السيستم: ${sys}`, st, act]);
        alert('تم الحفظ وإرسال البيانات إلى Google Sheets بنجاح!');
    }

    function addExpense() {
        let date = document.getElementById('expDate').value;
        let amt = parseFloat(document.getElementById('expAmt').value);
        let desc = document.getElementById('expDesc').value.trim();
        if(isNaN(amt) || amt <= 0 || !desc) return alert('أدخل بيانات المصروف بشكل صحيح');

        syncDataToSheet('Expenses', [date, amt, desc]);
        syncDataToSheet('Records', [date, 'مصروف نقدي', 'مصروفات', 'مصروف', desc, 'منصرف', amt]);
        alert('تم خصم المصروف وإرساله إلى Google Sheets بنجاح!');
    }
